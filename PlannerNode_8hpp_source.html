243template<typename NodeT>

246 const std::string & plugin_name)
247{
248 declare_parameter_if_not_declared(node, plugin_name + ".plugin", rclcpp::ParameterValue(""));
249 std::string plugin_type;
250 if (!node->get_parameter(plugin_name + ".plugin", plugin_type)) {
251 RCLCPP_FATAL(node->get_logger(), "'plugin' param not defined for %s", plugin_name.c_str());
